Step 2 — Create `scripts/ext01_diff_stage.py`. The harness performs three jobs: (1) load each dataset (text path + repository path); (2) compute `S_regex[ds][comp]` via the `s_linker13._has_standalone_mention` regex; (3) compute `S_v[ds][comp]` for each sub-variant by running its `_compute_standalone_mention_map` (LLM-backed; cache aggressively per the existing `PHASE_CACHE_DIR` convention so reruns are cheap); (4) roll up metrics; (5) apply the catastrophic-diff drop rule; (6) emit JSON.

Skeleton:

```python
#!/usr/bin/env python3
"""EXT-01 D-02 offline diff stage.

For each of the 5 benchmark datasets and each candidate sub-variant
(s_linker13g_pre, s_linker13g_sem), compute:
  - regex baseline anchor set: S_regex[ds][comp] = set of snums where
    s_linker13._has_standalone_mention(comp_name, sent.text) is True
  - variant anchor set: S_v[ds][comp] = set of snums where the variant's
    _compute_standalone_mention_map returns True

Roll up per-(variant, ds): min_jaccard_per_comp, mean_jaccard_weighted_by_|S_regex|,
count_components_with_J<0.5, max_symmetric_diff.

Apply the catastrophic-diff drop rule:
  Drop variant if (on TM or BBB):
    - min_jaccard_per_comp < 0.3, OR
    - max_symmetric_diff > 10, OR
    - count_components_with_J<0.5 > 25% of components.

Emit:
  - results/ablation_results/ablation_ext01_diff.json (machine-readable matrix)
  - .planning/phases/06-.../06-DIFF-MATRIX.md (human-readable report — written by Task 2)

This script does NOT invoke the full pipeline. It only runs Tier-1's
_compute_standalone_mention_map per variant + the regex baseline.
"""

from __future__ import annotations

import argparse
import json
import os
import sys
import time
from pathlib import Path
from typing import Any

# Repo root inferred from script location.
REPO_ROOT = Path(__file__).resolve().parent.parent
sys.path.insert(0, str(REPO_ROOT / "src"))

from llm_sad_sam.linkers.experimental.s_linker13 import SLinker13
from llm_sad_sam.linkers.experimental.s_linker13g_pre import SLinker13gPre
from llm_sad_sam.linkers.experimental.s_linker13g_sem import SLinker13gSem
from llm_sad_sam.core.document_loader_v2 import load_sentences, build_sent_map
from llm_sad_sam.pcm_parser_v2 import parse_pcm_repository


# Dataset table — paths must mirror run_ablation.py's dataset registry.
# If run_ablation.py exposes a DATASETS dict, prefer importing it. Otherwise
# fall back to the explicit list below (verify against ardoco/core/tests-base
# /src/main/resources/benchmark/<project>/).
DATASETS = {
    "mediastore":     ("benchmark/mediastore/<text>.txt",     "benchmark/mediastore/<repo>.repository"),
    "teastore":       ("benchmark/teastore/<text>.txt",       "benchmark/teastore/<repo>.repository"),
    "teammates":      ("benchmark/teammates/<text>.txt",      "benchmark/teammates/<repo>.repository"),
    "bigbluebutton":  ("benchmark/bigbluebutton/<text>.txt",  "benchmark/bigbluebutton/<repo>.repository"),
    "jabref":         ("benchmark/jabref/<text>.txt",         "benchmark/jabref/<repo>.repository"),
}
# IMPLEMENTATION NOTE: Resolve actual paths by replicating run_ablation.py's dataset registry.
# `grep -nE "mediastore|teastore|teammates|bigbluebutton|jabref" run_ablation.py` finds them.

HARD_TIER = {"teammates", "bigbluebutton"}


def compute_anchor_set(predicate, sentences, comp_names):
    """Returns dict[comp_name -> set[snum]] of snums where predicate(cname, sent.text) is True."""
    out = {c: set() for c in comp_names}
    for c in comp_names:
        for s in sentences:
            if predicate(c, s.text):
                out[c].add(s.number)
    return out


def regex_predicate(cname, text):
    """The frozen regex baseline — calls s_linker13's static method directly."""
    return SLinker13._has_standalone_mention(cname, text)


def variant_anchor_set(variant_inst, sentences, components):
    """Invokes the variant's Tier-1 _compute_standalone_mention_map and projects to set form."""
    smap = variant_inst._compute_standalone_mention_map(sentences, components)
    comp_names = variant_inst._get_comp_names(components) if hasattr(variant_inst, '_get_comp_names') \
                 else [c.component_name for c in components]
    out = {c: set() for c in comp_names}
    for (cname, snum), is_standalone in smap.items():
        if is_standalone and cname in out:
            out[cname].add(snum)
    return out


def jaccard(s_v: set, s_r: set) -> float:
    if not s_v and not s_r:
        return 1.0  # vacuous agreement
    union = s_v | s_r
    if not union:
        return 1.0
    return len(s_v & s_r) / len(union)


def symdiff(s_v: set, s_r: set) -> int:
    return len(s_v ^ s_r)


def rollup_dataset(per_comp_J: dict[str, float],
                   per_comp_D: dict[str, int],
                   per_comp_regex_size: dict[str, int]) -> dict[str, Any]:
    comps = list(per_comp_J.keys())
    if not comps:
        return {"min_jaccard_per_comp": 1.0, "mean_jaccard_weighted": 1.0,
                "count_components_with_J<0.5": 0, "max_symmetric_diff": 0,
                "n_components": 0}
    weights = [max(1, per_comp_regex_size[c]) for c in comps]
    weight_sum = sum(weights)
    return {
        "min_jaccard_per_comp": min(per_comp_J.values()),
        "mean_jaccard_weighted": sum(per_comp_J[c] * w for c, w in zip(comps, weights)) / weight_sum,
        "count_components_with_J<0.5": sum(1 for j in per_comp_J.values() if j < 0.5),
        "max_symmetric_diff": max(per_comp_D.values()),
        "n_components": len(comps),
    }


def apply_drop_rule(per_ds_rollup: dict[str, dict]) -> tuple[bool, list[str]]:
    """Returns (drop?, list_of_reasons). Reasons are human-readable for the report."""
    reasons = []
    for ds in HARD_TIER:
        r = per_ds_rollup.get(ds)
        if not r:
            continue
        if r["min_jaccard_per_comp"] < 0.3:
            reasons.append(f"{ds}: min_jaccard_per_comp={r['min_jaccard_per_comp']:.3f} < 0.3")
        if r["max_symmetric_diff"] > 10:
            reasons.append(f"{ds}: max_symmetric_diff={r['max_symmetric_diff']} > 10")
        pct = r["count_components_with_J<0.5"] / max(1, r["n_components"])
        if pct > 0.25:
            reasons.append(f"{ds}: count_components_with_J<0.5={r['count_components_with_J<0.5']}/"
                           f"{r['n_components']} = {pct:.1%} > 25%")
    return (len(reasons) > 0, reasons)


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--variants", nargs="+", default=["s_linker13g_pre", "s_linker13g_sem"])
    ap.add_argument("--datasets", nargs="+", default=list(DATASETS.keys()))
    ap.add_argument("--output", default="results/ablation_results/ablation_ext01_diff.json")
    args = ap.parse_args()

    variant_classes = {
        "s_linker13g_pre": SLinker13gPre,
        "s_linker13g_sem": SLinker13gSem,
    }

    matrix: dict[str, Any] = {
        "generated_at": time.strftime("%Y-%m-%dT%H:%M:%S"),
        "variants": {},
    }

    for vname in args.variants:
        cls = variant_classes[vname]
        per_ds: dict[str, Any] = {}
        for ds in args.datasets:
            text_path_rel, repo_path_rel = DATASETS[ds]
            # TODO: resolve the actual absolute paths via the same logic run_ablation.py uses.
            text_path, repo_path = _resolve_paths(text_path_rel, repo_path_rel)

            sentences = load_sentences(text_path)
            components = parse_pcm_repository(repo_path)
            comp_names = [c.component_name for c in components]

            # Regex baseline
            S_r_by_comp = compute_anchor_set(regex_predicate, sentences, comp_names)
            # Variant
            inst = cls(text_path=text_path, model_path=repo_path)
            S_v_by_comp = variant_anchor_set(inst, sentences, components)

            per_comp_J = {c: jaccard(S_v_by_comp.get(c, set()), S_r_by_comp.get(c, set()))
                          for c in comp_names}
            per_comp_D = {c: symdiff(S_v_by_comp.get(c, set()), S_r_by_comp.get(c, set()))
                          for c in comp_names}
            per_comp_regex_size = {c: len(S_r_by_comp.get(c, set())) for c in comp_names}

            per_ds[ds] = {
                "per_component": {c: {"J": per_comp_J[c], "D": per_comp_D[c],
                                       "|S_regex|": per_comp_regex_size[c],
                                       "|S_variant|": len(S_v_by_comp.get(c, set()))}
                                  for c in comp_names},
                "rollup": rollup_dataset(per_comp_J, per_comp_D, per_comp_regex_size),
            }

        drop, reasons = apply_drop_rule({ds: per_ds[ds]["rollup"] for ds in per_ds})
        matrix["variants"][vname] = {
            "per_dataset": per_ds,
            "drop": drop,
            "drop_reasons": reasons,
        }
        print(f"[{vname}] drop={drop} reasons={reasons}")

    Path(args.output).parent.mkdir(parents=True, exist_ok=True)
    with open(args.output, "w") as f:
        json.dump(matrix, f, indent=2, default=list)
    print(f"Wrote {args.output}")


def _resolve_paths(text_rel: str, repo_rel: str) -> tuple[str, str]:
    """Resolve dataset paths relative to the ardoco benchmark root.

    Strategy: import or replicate run_ablation.py's path logic.
    """
    # Walk up to find the ardoco benchmark root. Adjust to match run_ablation.py.
    # ardoco/core/tests-base/src/main/resources/benchmark/<project>/<file>
    candidates = [
        REPO_ROOT.parent / "ardoco" / "core" / "tests-base" / "src" / "main" / "resources",
        REPO_ROOT / "benchmark",
    ]
    for root in candidates:
        full_text = root / text_rel
        full_repo = root / repo_rel
        if full_text.exists() and full_repo.exists():
            return str(full_text), str(full_repo)
    raise FileNotFoundError(f"Could not locate dataset files for {text_rel}")


if __name__ == "__main__":
    main()
```
