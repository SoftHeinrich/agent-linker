<!-- Alias-block formatting helper (NEW — add to each new linker file) -->

```python
def _format_alias_block(self) -> str:
    """Format the KNOWN ALIASES block for the standalone-mention prompt.
    Project-agnostic: echoes whatever doc_knowledge discovered, possibly empty."""
    if not self.doc_knowledge or not self.doc_knowledge.aliases:
        return "(none discovered)"
    lines = []
    for term, entry in self.doc_knowledge.aliases.items():
        lines.append(f"{term} -> {entry.component}")
    return "\n".join(lines)
```

<!-- Linkmap-block formatting helper (NEW — *_full variants ONLY) -->

```python
def _format_linkmap_block(self) -> str:
    """Format the RUNNING LINK MAP block from raw seed links.
    Project-agnostic: echoes whatever ILinker3 returned, possibly empty."""
    if not getattr(self, "_raw_seed_links", None):
        return "(none yet)"
    lines = []
    seen = set()
    for lk in sorted(self._raw_seed_links, key=lambda l: l.sentence_number):
        key = (lk.sentence_number, lk.component_name)
        if key in seen:
            continue
        seen.add(key)
        lines.append(f"S{lk.sentence_number}: {lk.component_name}")
    return "\n".join(lines)
```
