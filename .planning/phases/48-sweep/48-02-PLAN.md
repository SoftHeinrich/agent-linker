import json, glob, os

total_p, total_c = 0, 0
for f in call_files:
    data = json.load(open(f))
    p = sum(c.get('token_usage', {}).get('prompt_tokens', 0) or 0 for c in data if c.get('token_usage'))
    cc = sum(c.get('token_usage', {}).get('completion_tokens', 0) or 0 for c in data if c.get('token_usage'))

est = (total_p * 0.00003) + (total_c * 0.00006)

print('GATE-08:', 'PASS (<= \$20)' if est <= 20 else 'FAIL (> \$20)')
