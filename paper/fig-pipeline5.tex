    layer1/.style={phase, fill=blue!8},
    layer2/.style={phase, fill=orange!10},

% Layer 1: Three parallel analyses
\node[layer1] (model)   {Model\\Analysis};
\node[layer1, right=0.8cm of model] (doc) {Document\\Analysis};
\node[layer1, right=0.8cm of doc] (seed) {Seed\\Extraction};

% Layer 2: Three parallel agents
\node[layer2, below=1.4cm of model] (disambig) {Seed Ref.\\Disambig.};
\node[layer2, right=0.6cm of disambig] (contextual) {Contextual\\Ref.\ Disc.};
\node[layer2, right=0.6cm of contextual] (anaphoric) {Anaphoric\\Ref.\ Res.};

% Knowledge flow arrows (Layer 1 -> Layer 2)

\draw[arr] (model.south) -- (disambig.north west);

\end{tikzpicture}
\caption{Pipeline architecture as a staged DAG. Layer~1 runs three analyses
in parallel: model analysis classifies name ambiguity, document analysis
discovers aliases (including trailing-word references), and seed extraction
recovers initial explicit-reference links without knowledge dependencies.
Layer~2 runs three recovery agents in parallel once Layer~1 completes:
seed disambiguation filters seed links using the acquired knowledge,
contextual discovery recovers alias-based and functional references, and
anaphoric resolution resolves pronominal references.
Layer~3 deduplicates their output by first-seen priority.}
\label{fig:pipeline}
\end{figure*}
